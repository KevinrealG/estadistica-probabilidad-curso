```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm

# ==========================================================
# PASO CRÍTICO: INGRESA LOS ÚLTIMOS 4 DÍGITOS DE TU CÓDIGO
# ==========================================================
mi_codigo = 4829  # <-- Reemplaza con tus últimos 4 dígitos
np.random.seed(mi_codigo)

# Parámetros únicos de tu reactor industrial (en mililitros)
mu = 250 + (mi_codigo % 50)          # Media entre 250 y 299 ml
sigma = round(4 + (mi_codigo % 11) * 0.5, 1) # Desviación entre 4.0 y 9.0 ml

# 1. Simulación de 1,000 mediciones continuas
datos_simulados = np.random.normal(loc=mu, scale=sigma, size=1000)

# 2. Verificación de la Regla Empírica (68 - 95 - 99.7) en tu muestra
dentro_1sigma = np.mean((datos_simulados >= mu - sigma) & (datos_simulados <= mu + sigma)) * 100
dentro_2sigma = np.mean((datos_simulados >= mu - 2*sigma) & (datos_simulados <= mu + 2*sigma)) * 100
dentro_3sigma = np.mean((datos_simulados >= mu - 3*sigma) & (datos_simulados <= mu + 3*sigma)) * 100

print(f"--- AUDITORÍA NORMAL DEL ANALISTA [{mi_codigo}] ---")
print(f"Parámetros Teóricos: Media (mu) = {mu} ml | Desviación (sigma) = {sigma} ml")
print(f"Regla Empírica Simulada: 1s = {dentro_1sigma:.1f}% | 2s = {dentro_2sigma:.1f}% | 3s = {dentro_3sigma:.1f}%")

# 3. Gráfico: Histograma Empírico + Curva de Densidad Teórica (PDF)
x = np.linspace(mu - 4*sigma, mu + 4*sigma, 200)
plt.figure(figsize=(8, 4))
plt.hist(datos_simulados, bins=30, density=True, alpha=0.6, color='skyblue', edgecolor='black', label='Simulación (n=1000)')
plt.plot(x, norm.pdf(x, mu, sigma), 'r-', lw=2.5, label=f'PDF Teórica N({mu}, {sigma})')
plt.axvline(mu, color='navy', linestyle='--', label='Media')
plt.title(f'Distribución Normal de Llenado - Analista {mi_codigo}')
plt.xlabel('Volumen (ml)')
plt.ylabel('Densidad de Probabilidad')
plt.legend()
plt.grid(alpha=0.3)
plt.show()

```
